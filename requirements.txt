import os
import json
import time
import re
import io
from pathlib import Path
import pandas as pd
from PIL import Image
from pydantic import BaseModel, Field
from google import genai
from google.genai import types
import pdf2image
import docx
import streamlit as st

# ---------------------------------------------------------------------------
# CONFIGURACIÓN DE PÁGINA STREAMLIT
# ---------------------------------------------------------------------------
st.set_page_config(
    page_title="Gestor Contable Automático",
    page_icon="📊",
    layout="wide"
)

# ---------------------------------------------------------------------------
# 1. ESQUEMA DE DATOS UNIFICADO
# ---------------------------------------------------------------------------
class ComprobanteData(BaseModel):
    tipo_movimiento: str = Field(description="Ingreso o Egreso segun la naturaleza de la transaccion")
    concepto_factura: str = Field(description="Numero de factura o concepto tomado del nombre del archivo o del cuerpo del comprobante")
    destinatario_remitente: str = Field(description="Nombre o identificacion de la persona o comercio receptor/emisor")
    monto_cop: float = Field(description="Valor numerico de la transaccion en COP sin simbolos de moneda ni puntos")
    fecha_hora: str = Field(description="Fecha y hora de la transaccion en formato YYYY-MM-DD HH:MM")
    referencia_operacion: str = Field(description="Numero de comprobante, autorizacion, referencia o ID de operacion")
    banco_plataforma: str = Field(description="Entidad financiera de origen/destino (ej. Nequi, Davivienda, Nu, Bancolombia, Bre-B)")
    medio_pago_tipo: str = Field(description="Tipo de movimiento (ej. Transferencia, QR, Llave, Pago de servicios)")

# ---------------------------------------------------------------------------
# 2. FUNCIONES NÚCLEO DE EXTRACCIÓN Y CONVERSIÓN
# ---------------------------------------------------------------------------
def procesar_comprobante(client: genai.Client, entrada_contenido, nombre_archivo: str, es_ingreso: bool = True, max_retries: int = 5) -> dict:
    prompt = f"""
    Analiza detalladamente este comprobante bancario, factura o documento contable.
    
    Información contextual importante:
    - Este archivo corresponde a un {'INGRESO' if es_ingreso else 'EGRESO'}.
    - El nombre original del archivo es: "{nombre_archivo}". 
    - Si es un INGRESO, el nombre del archivo suele ser el Número de Factura.
    - Si es un EGRESO, el nombre del archivo suele ser el Concepto del gasto.
    
    Extrae con precisión todos los campos solicitados según el esquema. 
    Estandariza las fechas al formato YYYY-MM-DD HH:MM.
    Extrae la cifra monetaria en valor numérico float puro sin puntos ni signo de pesos.
    """

    for intento in range(max_retries):
        try:
            response = client.models.generate_content(
                model='gemini-2.5-flash',
                contents=[entrada_contenido, prompt],
                config=types.GenerateContentConfig(
                    response_mime_type="application/json",
                    response_schema=ComprobanteData,
                    temperature=0.1
                ),
            )
            return json.loads(response.text)
        except Exception as e:
            error_msg = str(e)
            if ("429" in error_msg or "RESOURCE_EXHAUSTED" in error_msg or "503" in error_msg) and intento < max_retries - 1:
                match = re.search(r'retry in (\d+(\.\d+)?)s', error_msg, re.IGNORECASE)
                tiempo_espera = float(match.group(1)) + 2 if match else 35.0
                st.warning(f"⏳ Límite alcanzado. Esperando {tiempo_espera:.0f} segundos para reintentar...")
                time.sleep(tiempo_espera)
                continue
            raise e

def extraer_contenido_archivo(file_bytes: bytes, filename: str):
    """Convierte o extrae el contenido del archivo subido en una entrada válida para Gemini."""
    ext = Path(filename).suffix.lower()
    
    if ext in ['.png', '.jpg', '.jpeg', '.webp']:
        return [Image.open(io.BytesIO(file_bytes))]
    
    elif ext == '.pdf':
        paginas = pdf2image.convert_from_bytes(file_bytes, first_page=1, last_page=1)
        return paginas
    
    elif ext == '.docx':
        doc = docx.Document(io.BytesIO(file_bytes))
        texto_completo = "\n".join([p.text for p in doc.paragraphs if p.text.strip()])
        for tabla in doc.tables:
            for fila in tabla.rows:
                texto_completo += "\n" + " | ".join([celda.text.strip() for celda in fila.cells])
        return [texto_completo]
    
    return []

def generar_excel_en_memoria(df: pd.DataFrame) -> bytes:
    output = io.BytesIO()
    with pd.ExcelWriter(output, engine='openpyxl') as writer:
        df.to_excel(writer, sheet_name='Libro Auxiliar', index=False)
        worksheet = writer.sheets['Libro Auxiliar']
        
        # Ajustar ancho de columnas dinámicamente
        for col in worksheet.columns:
            max_len = max(len(str(cell.value or '')) for cell in col)
            col_letter = col[0].column_letter
            worksheet.column_dimensions[col_letter].width = max(max_len + 3, 14)
            
        # Formato de celda numérico en moneda COP
        for cell in worksheet['C'][1:]:
            cell.number_format = '$#,##0'
            
    return output.getvalue()

# ---------------------------------------------------------------------------
# 3. INTERFAZ DE USUARIO STREAMLIT
# ---------------------------------------------------------------------------
st.title("📊 Procesador Automático de Comprobantes Contables")
st.markdown("Carga comprobantes en formato **Imagen**, **PDF** o **Word** (.docx) para evaluarlos, editarlos y consolidarlos en Excel.")

with st.sidebar:
    st.header("⚙️ Configuración")
    api_key_input = st.text_input("Google Gemini API Key", type="password", value=os.getenv("GOOGLE_API_KEY", ""))
    st.info("Obtén tu API Key gratuita en Google AI Studio.")

col1, col2 = st.columns(2)

with col1:
    st.subheader("📥 Comprobantes de Ingreso")
    files_ingresos = st.file_uploader(
        "Archivos de Ingreso",
        type=['png', 'jpg', 'jpeg', 'webp', 'pdf', 'docx'],
        accept_multiple_files=True,
        key="ingresos"
    )

with col2:
    st.subheader("📤 Comprobantes de Egreso")
    files_egresos = st.file_uploader(
        "Archivos de Egreso",
        type=['png', 'jpg', 'jpeg', 'webp', 'pdf', 'docx'],
        accept_multiple_files=True,
        key="egresos"
    )

if st.button("🚀 Procesar Comprobantes", type="primary", use_container_width=True):
    if not api_key_input:
        st.error("❌ Ingresa tu API Key de Gemini en el panel lateral para continuar.")
    elif not files_ingresos and not files_egresos:
        st.warning("⚠️ Debes seleccionar al menos un archivo de Ingreso o Egreso.")
    else:
        client = genai.Client(api_key=api_key_input)
        registros = []
        
        archivos_totales = len(files_ingresos) + len(files_egresos)
        progreso_bar = st.progress(0)
        status_text = st.empty()
        contador = 0

        # Lote Ingresos
        for file in files_ingresos:
            contador += 1
            status_text.text(f"Procesando Ingreso [{contador}/{archivos_totales}]: {file.name}")
            try:
                contenidos = extraer_contenido_archivo(file.read(), file.name)
                for item in contenidos:
                    data = procesar_comprobante(client, item, nombre_archivo=Path(file.name).stem, es_ingreso=True)
                    registros.append(data)
                time.sleep(1.5)
            except Exception as e:
                st.error(f"Error procesando {file.name}: {e}")
            progreso_bar.progress(contador / archivos_totales)

        # Lote Egresos
        for file in files_egresos:
            contador += 1
            status_text.text(f"Procesando Egreso [{contador}/{archivos_totales}]: {file.name}")
            try:
                contenidos = extraer_contenido_archivo(file.read(), file.name)
                for item in contenidos:
                    data = procesar_comprobante(client, item, nombre_archivo=Path(file.name).stem, es_ingreso=False)
                    registros.append(data)
                time.sleep(1.5)
            except Exception as e:
                st.error(f"Error procesando {file.name}: {e}")
            progreso_bar.progress(contador / archivos_totales)

        status_text.text("✅ Procesamiento finalizado.")

        if registros:
            df = pd.DataFrame(registros)
            columnas_orden = [
                "tipo_movimiento", "concepto_factura", "monto_cop", 
                "fecha_hora", "referencia_operacion", "banco_plataforma", 
                "medio_pago_tipo", "destinatario_remitente"
            ]
            df = df[columnas_orden]
            df.columns = [
                "Tipo Movimiento", "Factura / Concepto", "Monto (COP)", 
                "Fecha / Hora", "Referencia", "Banco / Canal", 
                "Tipo Operación", "Destinatario / Remitente"
            ]
            st.session_state['df_resultados'] = df

# Mostrar tabla editable y botón de descarga si hay resultados
if 'df_resultados' in st.session_state:
    st.divider()
    st.subheader("✏️ Evaluación y Edición de Resultados")
    st.caption("Puedes hacer doble clic en cualquier celda para corregir los datos manualmente antes de descargar.")
    
    df_editado = st.data_editor(
        st.session_state['df_resultados'],
        num_rows="dynamic",
        use_container_width=True
    )

    excel_data = generar_excel_en_memoria(df_editado)

    st.download_button(
        label="📥 Descargar Reporte Consolidado en Excel",
        data=excel_data,
        file_name="Consolidado_Contable.xlsx",
        mime="application/vnd.openxmlformats-officedocument.spreadsheetml.sheet",
        use_container_width=True
    )
